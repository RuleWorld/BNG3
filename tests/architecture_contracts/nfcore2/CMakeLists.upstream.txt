project(NFcore2Tests CXX)

  test_exhaustive_matcher_matrix.cpp
  test_scaffold_reference_model.cpp
  test_scheduler_reference_model.cpp
  test_rule_family_stress.cpp
  test_dependency_exhaustive.cpp
  test_shadow_semantic_matrix.cpp
  test_graph_state_machine_extended.cpp
)
